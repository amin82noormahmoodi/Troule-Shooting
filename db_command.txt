-- Step 1: Create tables
DROP TABLE IF EXISTS solutions;
DROP TABLE IF EXISTS problems;
DROP TABLE IF EXISTS boards;

CREATE TABLE boards (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50) NOT NULL
);

CREATE TABLE problems (
    id SERIAL PRIMARY KEY,
    board_id INT REFERENCES boards(id) ON DELETE CASCADE,
    problem_text TEXT NOT NULL
);

CREATE TABLE solutions (
    id SERIAL PRIMARY KEY,
    problem_id INT REFERENCES problems(id) ON DELETE CASCADE,
    solution_text TEXT NOT NULL
);

-- Step 3: Generate random problems and solutions for each board
DO $$
DECLARE
    b RECORD;
    problem_count INT;
    solution_count INT;
    i INT;
    j INT;
    p_id INT;
BEGIN
    -- Loop over each board
    FOR b IN SELECT * FROM boards LOOP
        -- Random number of problems (between 2 and 5)
        problem_count := FLOOR(RANDOM() * 4 + 2);

        FOR i IN 1..problem_count LOOP
            INSERT INTO problems (board_id, problem_text)
            VALUES (b.id, FORMAT('Problem %s-%s: issue description for %s', b.name, i, b.name))
            RETURNING id INTO p_id;

            -- Random number of solutions (between 2 and 5)
            solution_count := FLOOR(RANDOM() * 4 + 2);

            FOR j IN 1..solution_count LOOP
                INSERT INTO solutions (problem_id, solution_text)
                VALUES (p_id, FORMAT('Solution %s-%s.%s: fix description for %s problem %s', 
                                     b.name, i, j, b.name, i));
            END LOOP;
        END LOOP;
    END LOOP;
END $$;

-- Step 4: Check the generated data
SELECT 
    b.name AS board_name,
    p.problem_text,
    s.solution_text
FROM boards b
JOIN problems p ON b.id = p.board_id
JOIN solutions s ON p.id = s.problem_id
ORDER BY b.id, p.id, s.id;
